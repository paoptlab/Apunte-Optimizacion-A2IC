\RequirePackage{import}

% El segundo argumento restaura las rutas de imágenes que cada documento
% configura suponiendo que la compilación comienza dentro de su subcarpeta.
\newcommand{\documentoOverleaf}[2]{%
  \AtBeginDocument{\graphicspath{{#2img/}{#2}}}%
  \import{#1}{main.tex}%
}
